from pathlib import Path
import re,json,ast
root=Path('SrvRestAstroLS_v1/backend/modules/telchi'); refs=json.loads(Path('/tmp/telchi-permission-refs.json').read_text())
for p in root.rglob('*'):
 if p.suffix not in {'.py','.sql','.md','.csv','.json','.yaml'} or p.stat().st_size>2_000_000:continue
 text=p.read_text(errors='replace')
 for n,line in enumerate(text.splitlines(),1):
  for pat in [r'\bTELCHI\s*\.\s*"?(@[A-Z0-9_]+|[A-Z][A-Z0-9]{2,5})\b',r'\b(?:FROM|JOIN)\s+([A-Z][A-Z0-9]{2,5})\b(?!\s*\.)']:
   for m in re.finditer(pat,line):
    t=m[1]
    if t in {'SELECT','VALUES','WHERE','TELCHI'}:continue
    refs.setdefault(t,[]).append(dict(path=str(p),line=n,text=line.strip()[:300]))
 if p.suffix=='.py':
  try:tree=ast.parse(text)
  except SyntaxError:continue
  for node in ast.walk(tree):
   if isinstance(node,(ast.Assign,ast.AnnAssign)) and node.value is not None:
    targets=node.targets if isinstance(node,ast.Assign) else [node.target]
    if not any(isinstance(t,ast.Name) and re.search(r'TABLES|table_names|DISCOVERY',t.id) for t in targets):continue
    for c in ast.walk(node.value):
     if isinstance(c,ast.Constant) and isinstance(c.value,str) and re.fullmatch(r'@[A-Z0-9_]+|[A-Z][A-Z0-9]{2,5}',c.value):
      refs.setdefault(c.value,[]).append(dict(path=str(p),line=c.lineno,text='table inventory declaration'))
for t in ['SYS','TELCHI']:refs.pop(t,None)
Path('/tmp/telchi-permission-refs.json').write_text(json.dumps(refs,indent=2))
print('Candidates',len(refs));print(' '.join(sorted(refs)))
